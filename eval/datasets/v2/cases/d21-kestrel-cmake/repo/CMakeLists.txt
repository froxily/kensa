cmake_minimum_required(VERSION 3.22)
project(Kestrel LANGUAGES CXX)
include(CheckCXXCompilerFlag)
check_cxx_compiler_flag("-Wall" HAS_WALL)
check_cxx_compiler_flag("-Wextra" HAS_WEXTRA)
check_cxx_compiler_flag("-Wpedantic" HAS_WPEDANTIC)
check_cxx_compiler_flag("-Wshadow" HAS_WSHADOW)
check_cxx_compiler_flag("-Wconversion" HAS_CONVERSION)
check_cxx_compiler_flag("-Wsign-conversion" HAS_SIGN_CONVERSION)
check_cxx_compiler_flag("-Wformat=2" HAS_FORMAT)
check_cxx_compiler_flag("-Wundef" HAS_UNDEF)
check_cxx_compiler_flag("-Wcast-align" HAS_CAST_ALIGN)
check_cxx_compiler_flag("-Wcast-qual" HAS_CAST_QUAL)
check_cxx_compiler_flag("-Woverloaded-virtual" HAS_OVERLOADED_VIRTUAL)
check_cxx_compiler_flag("-Wnon-virtual-dtor" HAS_NON_VIRTUAL_DTOR)
check_cxx_compiler_flag("-Wold-style-cast" HAS_OLD_STYLE_CAST)
check_cxx_compiler_flag("-Wdouble-promotion" HAS_DOUBLE_PROMOTION)
include(cmake/CompilerWarnings.cmake)
add_executable(kestrel src/main.cpp)
